Preparar los datos

# Crear una nueva variable para clasificar niños y adultos

datos$Child = ifelse(
  datos$Age < 12,
  "Niño",
  "Adulto"
)

# Convertir variables en factores

datos$Sex = as.factor(datos$Sex)
datos$Survived = as.factor(datos$Survived)
datos$Pclass = as.factor(datos$Pclass)

Boxplot con ggplot2

Distribución de edad por supervivencia

p1 = ggplot(
  datos,
  aes(
    x = Survived,
    y = Age,
    fill = Survived
  )
) +
  geom_boxplot(
    alpha = 0.7,
    outlier.color = "red",
    outlier.size = 2
  ) +
  scale_fill_manual(
    values = c(
      "0" = "#E74C3C",
      "1" = "lightgreen"
    ),
    labels = c(
      "0" = "No sobrevivió",
      "1" = "Sobrevivió"
    )
  ) +
  labs(
    title = "Distribución de edad por supervivencia",
    x = "Supervivencia",
    y = "Edad (Años)"
  ) +
  theme_minimal() +
  theme(
    legend.position = "none"
  )

print(p1)
